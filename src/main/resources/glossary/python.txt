WITH
Context manager for resource handling.

Example:
    with open("file.txt") as f:
        print(f.read())

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

YIELD
Pauses a generator function.

Example:
    def count():
        for i in range(5):
            yield i

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

ZIP
Combines two iterables.

Example:
    names = ["Alice", "Bob"]
    ages = [25, 30]
    for n, a in zip(names, ages):
        print(n, a)
